const SHEET_ID = 'COLE_SEU_ID_AQUI'; // <- ID da sua planilha const USERS = { 'rebeca.barbosa@justtraveltour.com': 'beca', 'taina.brasil@justtraveltour.com': 'taina' }; const NAMES = { 'beca': 'Rebeca', 'taina': 'Tainá' }; function issuesSheet(user) { return user + '_issues'; } function tasksSheet(user) { return user + '_tasks'; } function inboxSheet(user) { return user + '_inbox'; } function getUser(e) { try { return USERS[Session.getActiveUser().getEmail()] || null; } catch(err) { return null; } } function doGet(e) { const user = getUser(e); if (!user) return err('Usuário não autorizado'); const action = e.parameter.action; try { if (action === 'whoami') return ok({ user, name: NAMES[user] }); if (action === 'read') return ok(readSheet(resolveSheet(user, e.parameter.sheet))); return err('Ação inválida'); } catch(ex) { return err(ex.message); } } function doPost(e) { const user = getUser(e); if (!user) return err('Usuário não autorizado'); try { const body = JSON.parse(e.postData.contents); if (body.action === 'write') { writeSheet(resolveSheet(user, body.sheet), body.data); return ok('saved'); } if (body.action === 'send_demand') { const target = body.target; if (!target || target === user) return err('Destino inválido'); sendDemand(user, target, body.task); return ok('sent'); } if (body.action === 'accept_inbox') { acceptInbox(user, body.taskId); return ok('accepted'); } return err('Ação inválida'); } catch(ex) { return err(ex.message); } } function resolveSheet(user, key) { if (key === 'issues') return issuesSheet(user); if (key === 'tasks') return tasksSheet(user); if (key === 'inbox') return inboxSheet(user); return key; } function sendDemand(fromUser, toUser, task) { const rows = readSheet(inboxSheet(toUser)); rows.unshift(Object.assign({}, task, { id: Date.now().toString(), from: fromUser, fromName: NAMES[fromUser], receivedAt: new Date().toISOString(), accepted: false })); writeSheet(inboxSheet(toUser), rows); } function acceptInbox(user, taskId) { const all = readSheet(inboxSheet(user)); const item = all.find(r => r.id === taskId); writeSheet(inboxSheet(user), all.filter(r => r.id !== taskId)); if (item) { const tasks = readSheet(tasksSheet(user)); tasks.unshift(Object.assign({}, item, { id: Date.now().toString(), status: 'nao-iniciada', history: [], createdAt: new Date().toISOString() })); writeSheet(tasksSheet(user), tasks); } } function readSheet(sheetName) { const ss = SpreadsheetApp.openById(SHEET_ID); let sh = ss.getSheetByName(sheetName); if (!sh) { sh = ss.insertSheet(sheetName); return []; } const vals = sh.getDataRange().getValues(); if (vals.length <= 1) return []; return vals.slice(1).map(r => { try { return JSON.parse(r[0]); } catch(e) { return null; } }).filter(Boolean); } function writeSheet(sheetName, data) { const ss = SpreadsheetApp.openById(SHEET_ID); let sh = ss.getSheetByName(sheetName); if (!sh) sh = ss.insertSheet(sheetName); sh.clearContents(); sh.appendRow(['data']); data.forEach(item => sh.appendRow([JSON.stringify(item)])); } function ok(data) { return ContentService.createTextOutput(JSON.stringify({ ok: true, data })).setMimeType(ContentService.MimeType.JSON); } function err(msg) { return ContentService.createTextOutput(JSON.stringify({ ok: false, error: msg })).setMimeType(ContentService.MimeType.JSON); }
